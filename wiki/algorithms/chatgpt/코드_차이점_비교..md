import sys
sys.stdin = open('input.txt', 'r')
from collections import deque

T = int(input())

dx = [-1, 1, 0, 0]
dy = [0, 0, -1, 1]

def bfs(x, y) :
    q = deque()
    q.append((x, y))

    while q :
        x, y = q.popleft()

        for i in range(4) :
            nx = x + dx[i]
            ny = y + dy[i]

            if 0 <= nx < m and 0 <= ny < n and arr[nx][ny] == 1 and v[nx][ny] == 0:
                v[nx][ny] = 1
                q.append((nx, ny))
        
for _ in range(T) :
    m, n, k = map(int, input().split())
    arr = [[0] * n for _ in range(m)]
    v = [[0] * n for _ in range(m)]
    cnt = 0

    for _ in range(k) :
        x, y = map(int, input().split())
        arr[x][y] = 1

    for a in range(m) :
        for b in range(n) :
            if arr[a][b] == 1 and v[x][y] == 0 :
                bfs(a, b)
                cnt += 1

    print(cnt)

2번 코드:
import collections

t = int(input())

def bfs(x,y):
    queue = collections.deque()
    queue.append((x,y))
    dx, dy = [-1, 1, 0, 0], [0, 0, -1, 1]
    
    while queue:
        x, y = queue.popleft()
        
        # 주변 4방향을 모두 확인한다.
        for i in range(4):
            nx = dx[i] + x
            ny = dy[i] + y
            if 0 <= nx < n and 0 <= ny < m:
                if graph[nx][ny] == 1 and check[nx][ny] == -1:
                    check[nx][ny] = 1
                    queue.append((nx,ny))
    
for i in range(t):
    n, m, k = map(int, input().split())
    graph = [[0]*m for _ in range(n)]
    check = [[-1]*m for _ in range(n)]
    count = 0
    
    for j in range(k):
        x, y = map(int, input().split())
        graph[x][y] = 1
    
    # 배추흰지렁이(값이 1인곳)가 있는곳이고 아직 방문하지 않았을경우 bfs를 시작한다.
    for x in range(n):
        for y in range(m):
            if graph[x][y] == 1 and check[x][y] == -1:
                bfs(x,y)
                count += 1
    print(count)

### 💡 답변 (Assistant)
두 코드는 비슷한 기능을 하는 것 같네요. 하지만 몇 가지 다른 점이 있어요.

1. **라이브러리 import 방식:**
    - 첫 번째 코드는 `sys.stdin`을 파일로부터 입력을 받도록 설정하고 `deque`를 import 합니다.
    - 두 번째 코드는 `collections` 라이브러리를 전체 import 합니다.

2. **함수 내 변수 사용:**
    - 첫 번째 코드에서는 `bfs()` 함수 내부에서 `v[x][y]`를 확인하고 있지만, 실제로 사용되어야 하는 변수는 `v[a][b]`일 것 같아요.
    - 두 번째 코드에서는 `check` 배열을 활용하고 있습니다.

3. **코드 구성:**
    - 첫 번째 코드에서는 입력값을 받는 방식이 파일로부터 입력을 받고 있습니다.
    - 두 번째 코드는 직접적으로 입력을 받고 있습니다.

4. **변수명:**
    - 변수명에도 약간의 차이가 있어요. 첫 번째 코드에서는 `cnt`, `arr`, `v`를 사용하고, 두 번째 코드에서는 `count`, `graph`, `check`를 사용하고 있습니다.

코드의 핵심 로직인 BFS 알고리즘 자체는 두 코드에서 거의 유사합니다. 입력 및 출력 부분과 변수명, 일부 변수 사용 방식에서 차이가 있지만, 기능적으로 큰 차이가 없어 보입니다.

위의 가능성을 고려해본 후에도 결과 값이 다르다면, 코드의 각 부분에서 어떤 값을 얻고 있는지, 각 단계에서 변수들이 어떻게 변하는지를 출력하여 비교해보는 것이 도움이 될 수 있습니다. 코드 실행 중간에 추가적인 출력을 넣어서 각 단계별로 어떤 일이 일어나는지 확인해보세요. 이를 통해 코드의 어떤 부분에서 다른 결과가 나오고 있는지 더 잘 이해할 수 있을 것입니다.
