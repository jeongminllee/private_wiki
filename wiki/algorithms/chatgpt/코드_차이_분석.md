### ❓ 질문 (User)
```
from collections import deque

# N: 교실 크기, T: 시뮬레이션할 일수
N, T = map(int, input().split())

# F: 각 학생이 신봉하는 음식을 저장하는 2차원 배열
# 음식 종류를 비트마스크로 표현하여 조합(예: 민트초코)을 쉽게 처리합니다.
# M(우유) = 1 (001), C(초코) = 2 (010), T(민트) = 4 (100)
# 예: 민트초코 = 4 | 2 = 6 (110), 민트초코우유 = 4 | 2 | 1 = 7 (111)
F = [[0] * N for _ in range(N)]
for i in range(N):
    foods = input()
    for j in range(N):
        if foods[j] == 'T':
            F[i][j] = 4  # 민트
        elif foods[j] == 'C':
            F[i][j] = 2  # 초코
        else:
            F[i][j] = 1  # 우유

# B: 각 학생의 초기 신앙심을 저장하는 2차원 배열
B = [
    list(map(int, input().split()))
    for _ in range(N)
]

# (x, y) 좌표가 격자(교실) 범위 내에 있는지 확인하는 함수
def is_range(x, y):
    return 0 <= x < N and 0 <= y < N

# 상, 하, 좌, 우 4방향을 나타내는 변수 (전파 방향 결정에 사용)
dxy = [(-1, 0), (1, 0), (0, -1), (0, 1)] # 위, 아래, 왼쪽, 오른쪽

# 점심시간: 그룹의 대표를 찾고 신앙심을 합치는 과정을 처리하는 함수
def get_head_and_add(visited, x, y):
    # BFS(너비 우선 탐색)를 위한 큐 생성
    q = deque()
    q.append((x, y))
    visited[x][y] = True  # 방문 처리
    
    cnt = 1  # 그룹원 수 (시작 지점 포함)
    hx, hy = x, y  # 그룹의 대표자 후보 (초기값: 시작 학생)

    # BFS를 통해 같은 음식을 신봉하는 인접 학생 그룹을 모두 찾음
    while q:
        cx, cy = q.popleft()
        for dx, dy in dxy:
            nx, ny = cx + dx, cy + dy
            # 격자 밖이거나, 이미 방문했거나, 음식이 다르면 그룹원이 아님
            if not is_range(nx, ny) or visited[nx][ny] or F[cx][cy] != F[nx][ny]:
                continue
            
            # 같은 그룹에 속하므로 큐에 추가하고 방문 처리
            q.append((nx, ny))
            visited[nx][ny] = True
            cnt += 1  # 그룹원 수 증가

            # 대표자 선정 기준에 따라 대표자를 갱신
            # 1. 신앙심 B가 가장 큰 사람
            # 2. 신앙심이 같으면 행(r)이 가장 작은 사람
            # 3. 행도 같으면 열(c)이 가장 작은 사람
            # 튜플 비교: (신앙심, -행, -열)이 클수록 우선순위가 높음
            if (B[hx][hy], -hx, -hy) <= (B[nx][ny], -nx, -ny):
                hx, hy = nx, ny
    
    # [코드 핵심] 대표자의 신앙심을 (그룹원 수)만큼 증가시킵니다.
    # 이는 문제의 '아침'과 '점심' 규칙을 한번에 계산하기 위한 간소화입니다.
    # 원래 규칙: (아침) 모두 +1 -> (점심) 대표자에게 (cnt-1)만큼 신앙심 이전
    #   -> 대표자 최종 신앙심 = B_기존 + 1(아침) + (cnt-1)(점심) = B_기존 + cnt
    # 이 코드는 대표자의 최종 신앙심 결과만 계산하고, 다른 그룹원의 신앙심은 변경하지 않습니다.
    # 저녁 시간에는 대표자만 활동하므로, 이 계산만으로도 문제 해결이 가능합니다.
    B[hx][hy] += cnt
    
    # 그룹의 대표자 좌표 반환
    return hx, hy


# 점심 시간 전체 과정을 처리하는 함수
def step2():
    # step2가 실행되는 동안 방문 여부를 체크하는 배열
    visited = [[False] * N for _ in range(N)]
    
    heads = [] # 각 그룹의 대표자들을 저장할 리스트
    for i in range(N):
        for j in range(N):
            # 아직 그룹에 속하지 않은 학생이라면 새로운 그룹 탐색 시작
            if not visited[i][j]:
                # 그룹 탐색, 대표자 선정, 신앙심 정산 후 대표자 정보 받아오기
                head = get_head_and_add(visited, i, j)
                heads.append(head)
    
    # 모든 그룹의 대표자 리스트를 반환
    return heads

# 신봉하는 기본 음식의 개수를 반환하는 함수 (저녁 시간 전파 순서 정렬에 사용)
def get_food_count(food):
    if food in [1, 2, 4]: return 1 # 단일 음식
    if food in [3, 5, 6]: return 2 # 이중 조합
    return 3                      # 삼중 조합

# 저녁시간: 대표자들이 신앙을 전파하는 과정을 처리하는 함수
def step3(heads):
    # 전파 순서에 맞게 대표자들을 정렬하는 기준 함수
    def get_key(head):
        x, y = head
        # 1. 조합된 음식 개수 (단일 -> 이중 -> 삼중)
        # 2. 신앙심 높은 순 (-B)
        # 3. 행 번호 작은 순 (x)
        # 4. 열 번호 작은 순 (y)
        return (get_food_count(F[x][y]), -B[x][y], x, y)
    heads.sort(key=get_key)

    # 그날 저녁에 전파를 당한 학생은 '방어상태'가 됨 (전파 불가)
    is_defend = [[False] * N for _ in range(N)]

    # 정렬된 순서대로 대표자들이 전파 시작
    for x, y in heads:
        # 만약 해당 대표자가 이미 전파를 당해 방어상태가 되었다면 전파 불가
        if is_defend[x][y]:
            continue
        
        # 간절함(X) = 신앙심 - 1
        X = B[x][y] - 1
        # 전파 방향은 '전파 전' 신앙심을 기준으로 결정됨 (B[x][y]%4)
        dx, dy = dxy[B[x][y] % 4]
        # 전파자는 신앙심 1만 남김
        B[x][y] = 1
        food = F[x][y]

        # 간절함이 0이 되거나 격자 밖으로 나갈 때까지 전파
        while True:
            x, y = x + dx, y + dy # 전파 방향으로 한 칸 이동
            # 격자 밖으로 나가거나 간절함이 0이 되면 전파 종료
            if not is_range(x, y) or X <= 0:
                break
            # 전파 대상이 자신과 같은 음식을 신봉하면 전파하지 않음
            if food == F[x][y]:
                continue
            
            # 전파 대상은 그 즉시 방어 상태가 됨
            is_defend[x][y] = True
            
            # 강한 전파: 간절함(X) > 대상의 신앙심(B[x][y])
            if X > B[x][y]:
                F[x][y] = food      # 대상이 전파자의 사상에 완전히 동화됨
                X -= (B[x][y] + 1)  # 간절함 감소
                B[x][y] += 1        # 대상의 신앙심 1 증가
            # 약한 전파: 간절함(X) <= 대상의 신앙심(B[x][y])
            else:
                # 비트 OR(|) 연산으로 대상의 음식에 전파자의 음식을 추가
                F[x][y] |= food
                B[x][y] += X  # 대상의 신앙심이 남은 간절함만큼 증가
                break         # 약한 전파 후에는 간절함이 0이 되어 전파 종료

# 하루가 끝난 후, 각 음식별 신앙심 총합을 출력하는 함수
def print_all():
    S = [0] * 8 # 각 음식(1~7)의 신앙심 총합을 저장할 배열
    for i in range(N):
        for j in range(N):
            S[F[i][j]] += B[i][j]
    
    # 문제에서 요구하는 출력 순서에 맞게 출력
    # 민트초코우유(7), 민트초코(6), 민트우유(5), 초코우유(3), 우유(1), 초코(2), 민트(4)
    print(S[7], S[6], S[5], S[3], S[1], S[2], S[4])

# T일 동안 시뮬레이션 실행
for _ in range(T):
    # step 1 (아침)은 step 2에 통합되어 있음
    heads = step2() # step 2: 점심
    step3(heads)    # step 3: 저녁
    print_all()     # 결과 출력
```

```
from collections import deque

# 위 아래 왼쪽 오른쪽
di = [-1, 1, 0, 0]
dj = [0, 0, -1, 1]

N, T = map(int, input().split())

f_arr = [list(input()) for _ in range(N)]
n_arr = [list(map(int, input().split())) for _ in range(N)]

dct = {'M' : 1, 'C' : 2, 'T' : 4}   # 비트 001, 010, 100

# f_arr 을 문자 -> 숫자(bit)로 바꿈
for i in range(N) :
    for j in range(N) :
        f_arr[i][j] = dct[f_arr[i][j]]


def is_valid(i, j) :
    return 0 <= i < N and 0 <= j < N

# 점심 함수. 대표자 찾기
def noon(si, sj) :
    q = deque()
    visited[si][sj] = 1
    
    cnt = 1
    ri, rj = si, sj # 그룹의 대표자 후보 (초기값 설정)

    while q :
        ci, cj = q.popleft()

        if (n_arr[ri][rj], -ri, -rj) <= (n_arr[ci][cj], -ci, -cj) :
            ri, rj = ci, cj

        for d in range(4) :
            ni, nj = ci + di[d], cj + dj[d]

            if is_valid(ni, nj) and visited[ni][nj] == 0 and f_arr[ni][nj] == f_arr[ri][rj] :
                q.append((ni, nj))
                visited[ni][nj] = 1
                cnt += 1

    n_arr[ri][rj] += cnt

    return ri, rj

# 저녁 우선순위 정하기
def dinner_priority(meal) :
    if meal in {1, 2, 4} :  # 단일 음식
        return 1
    elif meal in {3, 5, 6} :    # 이중 음식
        return 2
    else :          # 삼중음식
        return 3
    
# 1. 아침 - 점심시간
# 아침 : 모든 학생 신앙심 + 1
for _ in range(T) :
    # 아침 겸 점심 
    # 인접한 학생들과 신봉 음식이 완전히 같은 경우에만 그룹을 형성.
    # 아침 + 점심 => 대표자 += 그룹원, 그룹원은 그대로

    # 그룹 내 대표자 1명 선정
    # 우선순위 : 신앙심 > i > j
    # bfs를 돌려서 대표자 리스트에 넣자. 
    rep_list = []
    visited = [[0] * N for _ in range(N)]
    for i in range(N) :
        for j in range(N) :
            if visited[i][j] == 0 :
                ri, rj = noon(i, j)
                rep_list.append((ri, rj))

    # 저녁
    # 전파 = 그룹 순대로
    # 민트, 초코, 우유, 초코우유, 민트우유, 민트초코, 민트초코우유
    # [4, 2, 1], [3, 5, 6], [7]
    # 같은 그룹 내에서는 대표자의 신앙심 > 행 > 열
    
    # 전파자는 신앙심 중 1만 남기고 간절함 = 신앙심 - 1 로 바꿔 전파에 사용
    # 전파 방향은 B % 4 에 의해 결정. 0, 1, 2, 3
    # 전파할 방향으로 한 칸씩 이동하면서 전파를 시도. 격자 밖으로 나가거나 간절함이 0이 되면 종료
    # 같은 음식을 신봉하는 경우, 전파하지 않고 다음 진행
    # 다른 음식을 신봉하는 경우, 전파 진행.
    # 전파자 > 대상 : 강한 전파
    # 전파자는 간절함이 대상 + 1 만큼 깎이며, 전파 대상의 신앙심은 1 증가, 대상의 신봉 음식 = 전파자 음식
    # 이때 간절함이 0이 되면 더 이상 전파를 진행하지 않고 종료.
    # 전파자 <= 대상 : 약한 전파
    # 대상은 전파자의 음식에도 관심을 가짐.
    # 전파자는 간절함이 0이 되고, 대상의 신앙심은 간절함만큼 증가

    dinner_list = [[] for _ in range(4)]

    for ri, rj in rep_list :
        priority = dinner_priority(f_arr[ri][rj])
        dinner_list[priority].append([-n_arr[ri][rj], ri, rj])
        
    for i in range(1, 4) :
        dinner_list[i].sort()

    is_defend = [[0] * N for _ in range(N)]
    
    for dinner in dinner_list[1:] :
        if len(dinner) == 0 :
            continue

        for _, ri, rj in dinner :
            if is_defend[ri][rj] :
                continue

            score = n_arr[ri][rj] - 1
            n_arr[ri][rj] = 1

            dr = score % 4
            food = f_arr[ri][rj]

            while True :
                ri, rj = ri + di[dr], rj + dj[dr]

                if not is_valid(ri, rj) or score <= 0 :
                    break

                if food == f_arr[ri][rj] :
                    continue
                
                # 전파 대상은 방어 상태
                is_defend[ri][rj] = 1

                if score > n_arr[ri][rj] :
                    f_arr[ri][rj] = food
                    n_arr[ri][rj] += 1
                    score -= n_arr[ri][rj]
                
                else :
                    f_arr[ri][rj] |= food
                    n_arr[ri][rj] += score



    food_list = [0] * 8
    for i in range(N) :
        for j in range(N) :
            food_list[f_arr[i][j]] += n_arr[i][j]

    # dct = {'M' : 1, 'C' : 2, 'T' : 4}
    for i in (7, 6, 5, 3, 1, 2, 4) :
        print(food_list[i], end=' ')
    print()
```

## 핵심 차이 요약

| 구분 | 첫 번째 코드 | 두 번째 코드 | 영향 |
|---|---|---|---|
| 점심 BFS 시작 | 시작 좌표를 큐에 삽입 | 큐에 시작 좌표를 넣지 않음 | 그룹 탐색 자체가 실행되지 않음 |
| 전파 방향 | 전파 전 신앙심 `% 4` | 간절함 `(신앙심 - 1) % 4` | 모든 전파 방향이 달라짐 |
| 약한 전파 종료 | 전파 후 `break` | `break` 없음 | 약한 전파 후에도 계속 전파함 |
| 강한 전파 계산 | 대상 기존 신앙심을 기준으로 차감 | 증가 후 차감 | 결과는 동일 |
| 저녁 정렬 | 전체 대표자 리스트 정렬 | 음식 개수별 버킷 정렬 | 결과는 동일 |
| 출력 형식 | 마지막 숫자 뒤 공백 없음 | 마지막 숫자 뒤 공백 있음 | 일반적으로 채점 영향 없음 |

# 1. 두 번째 코드의 BFS가 시작되지 않음

첫 번째 코드는 큐에 시작 좌표를 넣습니다.

```python
q = deque()
q.append((x, y))
visited[x][y] = True
```

그러나 두 번째 코드는 다음과 같습니다.

```python
def noon(si, sj):
    q = deque()
    visited[si][sj] = 1
```

`q.append((si, sj))`가 없습니다.

따라서 바로 다음의 반복문이 한 번도 실행되지 않습니다.

```python
while q:
    ci, cj = q.popleft()
```

그 결과 두 번째 코드에서는 모든 학생이 항상 크기 1인 그룹으로 처리됩니다.

```python
cnt = 1
ri, rj = si, sj
n_arr[ri][rj] += cnt
```

즉, 실제로 인접한 학생들이 같은 음식을 신봉하더라도 그룹이 형성되지 않습니다.

## 수정

```python
def noon(si, sj):
    q = deque()
    q.append((si, sj))  # 반드시 추가
    visited[si][sj] = 1
```

이 차이 하나만으로도 두 코드의 점심 결과는 완전히 달라집니다.

# 2. 전파 방향을 계산하는 기준이 다름

첫 번째 코드는 전파하기 전 신앙심으로 방향을 정합니다.

```python
X = B[x][y] - 1

dx, dy = dxy[B[x][y] % 4]

B[x][y] = 1
```

처리 순서는 다음과 같습니다.

1. 간절함은 `신앙심 - 1`
2. 방향은 기존 신앙심 `% 4`
3. 전파자의 신앙심을 1로 변경

반면 두 번째 코드는 다음과 같습니다.

```python
score = n_arr[ri][rj] - 1
n_arr[ri][rj] = 1

dr = score % 4
```

`score`는 이미 `기존 신앙심 - 1`이므로, 방향을 다음과 같이 계산하는 셈입니다.

```python
(기존 신앙심 - 1) % 4
```

하지만 올바른 방향은 다음입니다.

```python
기존 신앙심 % 4
```

예를 들어 대표자의 신앙심이 `5`라면:

```python
# 올바른 방향
5 % 4 == 1  # 아래

# 두 번째 코드
(5 - 1) % 4 == 0  # 위
```

방향이 완전히 달라집니다.

## 수정 방법 1: 신앙심을 1로 바꾸기 전에 방향 저장

```python
dr = n_arr[ri][rj] % 4
score = n_arr[ri][rj] - 1
n_arr[ri][rj] = 1
```

## 수정 방법 2: score를 이용한다면 1을 다시 더하기

```python
score = n_arr[ri][rj] - 1
dr = (score + 1) % 4
n_arr[ri][rj] = 1
```

첫 번째 방법이 의미가 더 명확합니다.

# 3. 약한 전파 후 종료하지 않음

첫 번째 코드는 약한 전파가 발생하면 즉시 전파를 종료합니다.

```python
else:
    F[x][y] |= food
    B[x][y] += X
    break
```

약한 전파에서는 남은 간절함을 모두 대상에게 사용하므로, 이후 전파를 계속할 수 없습니다.

반면 두 번째 코드는 다음과 같습니다.

```python
else:
    f_arr[ri][rj] |= food
    n_arr[ri][rj] += score
```

여기에 `break`가 없습니다.

더 큰 문제는 `score`도 0으로 만들지 않는다는 점입니다. 따라서 두 번째 코드는 약한 전파를 한 후에도 같은 `score`를 가지고 다음 칸으로 계속 이동합니다.

예를 들어 간절함이 3이고 첫 번째 대상에게 약한 전파가 발생하면:

```python
대상1 신앙심 += 3
```

정상적으로는 여기서 종료되어야 합니다. 그러나 두 번째 코드는 계속 이동해서 다른 대상에게도 다시:

```python
대상2 신앙심 += 3
대상3 신앙심 += 3
```

과 같이 전파할 수 있습니다.

## 수정

```python
else:
    f_arr[ri][rj] |= food
    n_arr[ri][rj] += score
    break
```

또는 다음처럼 작성할 수도 있지만, `break`가 더 직접적입니다.

```python
else:
    f_arr[ri][rj] |= food
    n_arr[ri][rj] += score
    score = 0
```

# 4. 강한 전파의 계산 순서는 다르지만 결과는 동일함

첫 번째 코드:

```python
if X > B[x][y]:
    F[x][y] = food
    X -= B[x][y] + 1
    B[x][y] += 1
```

두 번째 코드:

```python
if score > n_arr[ri][rj]:
    f_arr[ri][rj] = food
    n_arr[ri][rj] += 1
    score -= n_arr[ri][rj]
```

대상의 기존 신앙심을 `b`라고 하면 첫 번째 코드는:

```python
score -= b + 1
b += 1
```

두 번째 코드는:

```python
b += 1
score -= b
```

두 계산 모두 최종적으로 다음과 같습니다.

```python
대상 신앙심 = 기존 신앙심 + 1
간절함 = 기존 간절함 - (기존 대상 신앙심 + 1)
```

따라서 이 부분은 표현 순서만 다르고 결과는 같습니다.

다만 첫 번째 코드가 문제의 수식을 더 직접적으로 보여주므로 이해하기 쉽습니다.

# 5. 점심 그룹 탐색에서 음식 비교 대상이 다름

첫 번째 코드:

```python
if (
    not is_range(nx, ny)
    or visited[nx][ny]
    or F[cx][cy] != F[nx][ny]
):
    continue
```

현재 BFS 위치와 다음 위치의 음식이 같은지 비교합니다.

두 번째 코드:

```python
if (
    is_valid(ni, nj)
    and visited[ni][nj] == 0
    and f_arr[ni][nj] == f_arr[ri][rj]
):
```

다음 위치와 현재 대표자의 음식을 비교합니다.

정상적인 BFS에서는 같은 그룹 구성원 모두 같은 음식을 신봉하므로, 두 방식의 결과는 같습니다. 대표자가 바뀌더라도 대표자는 같은 그룹에 속하기 때문에 음식은 동일합니다.

다만 코드의 의도를 더 분명히 하려면 시작 음식이나 현재 위치를 기준으로 비교하는 편이 좋습니다.

```python
group_food = f_arr[si][sj]

if (
    is_valid(ni, nj)
    and not visited[ni][nj]
    and f_arr[ni][nj] == group_food
):
```

# 6. 대표자를 선정하는 시점이 다름

첫 번째 코드는 새로운 학생을 BFS 큐에 넣을 때 대표자를 비교합니다.

```python
q.append((nx, ny))
visited[nx][ny] = True
cnt += 1

if (B[hx][hy], -hx, -hy) <= (B[nx][ny], -nx, -ny):
    hx, hy = nx, ny
```

두 번째 코드는 큐에서 학생을 꺼냈을 때 비교합니다.

```python
ci, cj = q.popleft()

if (n_arr[ri][rj], -ri, -rj) <= (
    n_arr[ci][cj],
    -ci,
    -cj,
):
    ri, rj = ci, cj
```

두 번째 코드에 시작점 삽입만 제대로 추가한다면, 모든 그룹원이 결국 한 번씩 큐에서 나오므로 결과는 동일합니다.

```python
q.append((si, sj))
```

현재 두 번째 코드에서는 큐가 비어 있어서 대표자 비교도 전혀 실행되지 않습니다.

```python
dinner_list = [[] for _ in range(4)]

for ri, rj in rep_list:
```

```python
for dinner in dinner_list[1:]:
```

두 구현 모두 최종 정렬 기준은 다음과 같습니다.

# 두 번째 코드에서 반드시 고쳐야 하는 부분

최소한 다음 세 군데를 수정해야 첫 번째 코드와 같은 동작을 합니다.

```python
def noon(si, sj):
    q = deque()
    q.append((si, sj))  # 1. BFS 시작점 추가
    visited[si][sj] = 1

    cnt = 1
    ri, rj = si, sj

    while q:
        ci, cj = q.popleft()

        if (n_arr[ri][rj], -ri, -rj) <= (
            n_arr[ci][cj],
            -ci,
            -cj,
        ):
            ri, rj = ci, cj

        for d in range(4):
            ni = ci + di[d]
            nj = cj + dj[d]

            if (
                is_valid(ni, nj)
                and not visited[ni][nj]
                and f_arr[ni][nj] == f_arr[ci][cj]
            ):
                q.append((ni, nj))
                visited[ni][nj] = 1
                cnt += 1

    n_arr[ri][rj] += cnt
    return ri, rj
```

저녁 부분은 다음과 같이 수정해야 합니다.

```python
# 2. 전파 전 신앙심으로 방향 결정
dr = n_arr[ri][rj] % 4

score = n_arr[ri][rj] - 1
n_arr[ri][rj] = 1
food = f_arr[ri][rj]

while True:
    ri += di[dr]
    rj += dj[dr]

    if not is_valid(ri, rj) or score <= 0:
        break

    if food == f_arr[ri][rj]:
        continue

    is_defend[ri][rj] = 1

    if score > n_arr[ri][rj]:
        f_arr[ri][rj] = food
        score -= n_arr[ri][rj] + 1
        n_arr[ri][rj] += 1

    else:
        f_arr[ri][rj] |= food
        n_arr[ri][rj] += score
        break  # 3. 약한 전파 후 종료
```

결론적으로 두 번째 코드의 오답 가능성이 가장 높은 원인은 다음 세 줄입니다.

```python
# 빠져 있음
q.append((si, sj))

# 잘못된 방향
dr = score % 4

# 빠져 있음
break  # 약한 전파 직후
```

이 세 부분을 수정하면 두 코드의 핵심 로직은 사실상 동일해집니다.
