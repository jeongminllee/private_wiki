# ====================
"""
미생물 연구
N*N 정사각형 크기의 배지 

1. 미생물 투입
- 좌측 하단 좌표 (r1, c1), 우측 상단 좌표 (r2, c2) 직사각형 영역에 한 무리의 미생물 투입
- 만약 영역 내에 다른 미생물이 존재한다면 새로 투입된 미생물이 그 영역 내의 미생물들을 잡아먹음.
- 기존에 있던 어떤 미생물 무리 A가 새로 투입된 미생물 무리 B에게 잡아먹히면서 A무리가 차지한 영역이
    둘 이상 나눠지게 되면 나눠진 A무리의 미생물은 배지에서 모두 사라짐.

2. 배지 이동
- 모든 미생물을 새로운 배지로 이동. 새로운 배지의 크기도 기존 용기와 동일. 
- 이 과정은 기존 배지에 미생물이 한 마리도 존재하지 않을 떄 까지 다음 작업을 반복
- 기존 배지에 있는 무리 중 가장 차지한 영역이 넓은 무리를 하나 선택, 
- 만약 이런 미생물 무리가 둘 이상이면 먼저 투입된 미생물 선택
- 우선순위 : 크기 > idx
- 선택된 미생물 무리를 새 배지로 옮김. 
- 이 때, 무리는 기존 용기에서의 형태를 유지해야 함.
- 미생물 무리가 차지한 영역이 배양 용기의 범위를 벗어나지 않고, 
- 다른 미생물의 영역과 겹치지 않도록 두어야 함.
- 이 조건 안에서 최대한 x좌표가 작은 위치로 미생물을 옮겨야 함. 
- 그런 위치가 둘 이상일 경우 최대한 y좌표가 작은 위치로 옮겨야 함.
- 우선순위 : x좌표 > y좌표
- 만약 어떤 곳에도 둘 수 없는 미생물 무리가 발견되면, 그 미생물 무리는 사라짐.

3. 실험 결과 기록
- 미생물 무리 중 상하좌우로 맞닿은 면이 있는 무리끼리는 '인접한 무리' 라고 표현함.
- 모든 '인접한 무리' 쌍을 확인. 이때 두 무리 A와 B가 맞닿은 면이 둘 이상이더라도 (A, B) 쌍은 한 번만 확인
- 확인한 두 무리가 A, B라면 (A영역) * (B영역) 만큼의 성과를 얻음.
- 확인한 모든 쌍의 성과를 더한 값이 이 실험의 결과. 이 값을 기록

"""

from collections import deque

di = [-1, 1, 0, 0]
dj = [0, 0, -1, 1]

def microoranism_insert(maps: list[list[int]], lst:list[int], num:int) :
    r1, c1, r2, c2 = lst
    for i in range(c1, c2) :
        for j in range(r1, r2) :
            maps[i][j] = num

def group() -> list[int]:
    v = [[0] * N for _ in range(N)]
    glst = []

    for si in range(N) :
        for sj in range(N) :
            # 미방문 미생물이면 그룹
            if v[si][sj] == 0 and maps[si][sj] > 0 :
                tlst = bfs(v, si, sj)
                glst.append(tlst)

    return glst

def bfs(v:list[list[int]], si:int, sj:int) -> list[list[int, int]] :
    q = deque()
    start_num = maps[si][sj]
    res_lst = []

    q.append((si, sj))
    v[si][sj] = start_num
    res_lst.append(start_num)
    res_lst.append([si, sj])

    while q :
        ci, cj = q.popleft()
        # 네 방향, 범위 내, 미방문, 조건(같은 값이면)
        for d in range(4) :
            ni, nj = ci + di[d], cj + dj[d]
            if 0<=ni<N and 0<=nj<N and v[ni][nj] == 0 and maps[ni][nj] == start_num :
                q.append((ni, nj))
                v[ni][nj] = start_num
                res_lst.append([ni, nj])

    return res_lst  # [start_num, [i, j], [i, j], ...]

def del_sort(glst:list[list[int]]) -> list[list[int]] :
    del_set = set() # 삭제할 그룹 인덱스
    for i in range(len(glst) - 1) :
        for j in range(i+1, len(glst)) :    # 가능한 2개를 선택하는 모든 조합
            if glst[i][0] == glst[j][0] :   # 가은 미생물 번호를 가진 그룹
                del_set.add(i)
                del_set.add(j)

    for i in range(len(glst)-1, -1, -1) :   # 삭제는 뒤부터
        if i in del_set :
            glst.pop(i)                     # 삭제
    # 정렬(크기가 큰 순, 번호가 작은순)
    glst.sort(key=lambda x:(-len(x), x[0]))

    # 그룹 내 좌표를 (0, 0) 기준의 상대좌표로 저장
    for lst in glst :
        # i, j 최소값 구하기 => (si, sj) => 모든 좌표에서 -si, -sj
        si, sj = N, N
        for ci, cj in lst[1:] : # lst[0] 은 미생물 번호
            si = min(si, ci)
            sj = min(sj, cj)

        for idx in range(1, len(lst)) :     # 모든 좌표를 (0, 0) 기준 상대좌표로 저장
            lst[idx][0] -= si
            lst[idx][1] -= sj

    return glst

def move(glst:list[list[int]]) -> list[list[int]]:
    new_maps = [[0] * N for _ in range(N)]

    for lst in glst :   # 한 그룹씩 배치 가능한 최소 위치(si, sj)에 배치
        start = lst[0]
        si, sj = check(new_maps, lst[1:])   # 가능한 시작 좌표 리턴, 불가능한 경우 (-1, -1) 리턴
        if (si, sj) != (-1, -1) :
            for ci, cj in lst[1:] :
                new_maps[si+ci][sj+cj] = start

    return new_maps

def check(maps:list[list[int]], lst: list[int]) :
    # 최소열/최소행 기준으로 배치가능한지 확인 체크 후 기준좌표 리턴(불가능 시 (-1,-1))
    for sj in range(N) :
        for si in range(N) :    # 가능한 모든 기준위치
            # 기준 (si, sj)에 더한 모든 좌표가 범위내이고, 0이면 성공
            for ci, cj in lst :
                if si+ci >= N or sj+cj >= N or maps[si+ci][sj+cj] != 0 :
                    break

            else :      # 모든 좌표 배치 가능
                return si, sj
    # 가능한 기준좌표 없음
    return -1, -1

# 내 그룹개수를 세고(cnt), 인접한 다른 블럭들 카운트(cnts) 구해서 곱해서 누적
def recode_res(maps:list[list[int]]) :
    v = [[0] * N for _ in range(N)]
    res = 0

    for si in range(N) :
        for sj in range(N) :
            # 미방문 세포 만나면 점수 계산
            if v[si][sj] == 0 and maps[si][sj] > 0 :
                t = bfs_score(v, si, sj)
                res += t

    return res

def bfs_score(v:list[list[int]], si:int, sj:int) :
    q = deque()
    w = [[0]* N for _ in range(N)]
    cnts = []

    q.append((si, sj))
    v[si][sj] = 1
    cnt = 1

    while q :
        ci, cj = q.popleft()
        # 네방향, 범위내, 미방문
        for d in range(4) :
            ni, nj = ci + di[d], cj+dj[d]

            if 0<=ni<N and 0<=nj<N and v[ni][nj] == 0 :
                if maps[ni][nj] == maps[si][sj] :   # 같은 미생물 cnt 세기
                    q.append((ni, nj))
                    v[ni][nj] = 1
                    cnt += 1

                # 다른 미생물이면 첫 발견시 그 세포의 개수 세기 (방문 표시 w) 
                elif maps[ni][nj] > 0 and w[ni][nj] == 0 :
                    tcnt = bfs_adj(w, ni, nj)
                    cnts.append(tcnt)

    res = 0
    for other_cnt in cnts :
        res += cnt * other_cnt

    return res

def bfs_adj(v:list[list[int]], si:int, sj:int) :
    # 같은 값일 때 그 개수 세고 리턴
    q = deque()

    q.append((si, sj))
    v[si][sj] = 1
    cnt = 1

    while q :
        ci, cj = q.popleft()
        # 네방향, 범위내, 미방문, 조건(같은 값)
        for d in range(4) :
            ni, nj = ci + di[d], cj+dj[d]

            if 0<=ni<N and 0<=nj<N and v[ni][nj] == 0 and maps[si][sj] == maps[ni][nj]:
                q.append((ni, nj))
                v[ni][nj] = 1
                cnt += 1

    return cnt

if __name__ == "__main__" :
    N, Q = map(int, input().split())
    maps = [[0] * N for _ in range(N)]

    for test_num in range(1, Q+1) :
        input_lst = list(map(int, input().split())) # r1, c1, r2, c2

        # 1. 미생물 투입
        microoranism_insert(maps, input_lst, test_num)

        # 2. 배지 이동 glst = [미생물번호, [i1,j1],[i2,j2],...]
        glst = group()

        # 3. 쪼개진 개체 삭제 및 정렬 (크기, 오래된 : 미생물 번호가 작은)
        glst = del_sort(glst)

        # 4. 넓이가 큰, 오래된 순으로 정렬된 그룹 이동 (재배치)
        maps = move(glst)

        # 5. 결과 기록
        res = recode_res(maps)

        print(res)

오랜만에 알고리즘 풀러왔어. 이거 분석좀 해줘.

### 💡 답변 (Assistant)
전체적으로 **시뮬레이션 흐름은 잘 잡았어.**  
구조는 문제 요구사항이랑 거의 1:1로 대응돼.

1. `microoranism_insert()`  
   새 미생물 직사각형 투입, 기존 미생물 덮어쓰기

2. `group()` + `bfs()`  
   현재 배지에서 연결된 미생물 무리들을 찾음

3. `del_sort()`  
   같은 미생물 번호가 여러 덩어리로 쪼개졌으면 전부 삭제  
   이후 `크기 큰 순 -> 번호 작은 순` 정렬

4. `move()` + `check()`  
   새 배지에 우선순위대로 재배치

5. `recode_res()`  
   인접한 무리 쌍의 `면적 곱` 합산

핵심 로직은 꽤 괜찮아. 다만 아래 부분들은 조심해야 해.

**1. 좌표계 확인 필요**

너는 투입할 때 이렇게 하고 있어.

```python
for i in range(c1, c2):
    for j in range(r1, r2):
        maps[i][j] = num
```

즉 내부적으로는:

```python
maps[y][x]
```

처럼 쓰고 있어. 문제에서 `(r1, c1)`을 사실상 `(x, y)`처럼 준다면 이 방식이 맞아.  
하지만 문제에서 진짜 행/열 의미의 `(row, col)`이라면 뒤집힌 거야.

그리고 `check()`도:

```python
for sj in range(N):
    for si in range(N):
```

이렇게 되어 있어서 **x좌표 작은 순, 그다음 y좌표 작은 순**으로 탐색하고 있어.  
너의 좌표계가 `maps[y][x]`라면 이건 문제 조건에 맞는 방향이야.

**2. `del_sort()`는 의도는 맞음**

```python
if glst[i][0] == glst[j][0]:
    del_set.add(i)
    del_set.add(j)
```

같은 번호의 미생물이 여러 연결 컴포넌트로 나뉘면 모두 삭제하는 구조라서 문제 조건이랑 맞아.

다만 정렬은 이렇게 쓰는 게 더 명확해.

```python
glst.sort(key=lambda x: (-(len(x) - 1), x[0]))
```

현재 `len(x)`에는 미생물 번호 `x[0]`도 포함돼 있어서 실제 면적은 `len(x) - 1`이야. 정렬 결과는 같지만, 읽는 사람 입장에서는 `len(x)-1`이 훨씬 안전해 보여.

**3. 점수 계산은 동작할 가능성이 높지만 복잡함**

현재 `recode_res()`는 BFS로 한 무리를 잡고, 그 주변의 다른 무리를 만나면 `bfs_adj()`로 면적을 세는 방식이야.

이 방식은 `v`를 이용해서 이미 처리한 무리와의 쌍은 다시 안 세도록 되어 있어. 그래서 같은 쌍을 두 번 세는 문제는 어느 정도 막고 있어.

하지만 훨씬 간단하고 안전한 방식은 이거야.

```python
def record_res(board: list[list[int]]) -> int:
    area = {}

    for i in range(N):
        for j in range(N):
            if board[i][j] > 0:
                area[board[i][j]] = area.get(board[i][j], 0) + 1

    pairs = set()

    for i in range(N):
        for j in range(N):
            if board[i][j] == 0:
                continue

            for di, dj in [(1, 0), (0, 1)]:
                ni, nj = i + di, j + dj

                if 0 <= ni < N and 0 <= nj < N:
                    a = board[i][j]
                    b = board[ni][nj]

                    if b > 0 and a != b:
                        pairs.add(tuple(sorted((a, b))))

    return sum(area[a] * area[b] for a, b in pairs)
```

이 방식이 더 좋은 이유는:

- 인접 쌍을 `set`으로 직접 관리함
- 같은 쌍이 여러 면에서 맞닿아도 한 번만 계산됨
- BFS를 여러 번 돌리지 않음
- 코드가 문제 문장과 더 비슷함

**4. 타입 힌트 하나는 위험함**

```python
def bfs(v:list[list[int]], si:int, sj:int) -> list[list[int, int]]:
```

`list[int, int]` 형태는 타입 힌트로 부적절해. 게다가 반환값도 실제로는:

```python
[start_num, [i, j], [i, j], ...]
```

처럼 `int`와 `list[int]`가 섞여 있어.

차라리 간단히 이렇게 두는 게 나아.

```python
from typing import Any

def bfs(v: list[list[int]], si: int, sj: int) -> list[Any]:
```

더 깔끔하게 하려면 구조 자체를:

```python
[id, cells]
```

형태로 바꾸는 게 좋아.

```python
[num, [[i, j], [i, j], ...]]
```

지금처럼 `[num, [i,j], [i,j], ...]`도 가능하지만, 나중에 디버깅할 때 헷갈릴 수 있어.

**결론**

현재 코드는 큰 방향은 맞아. 특히 **투입 → 분리된 무리 삭제 → 크기/번호순 이동 → 인접 점수 계산** 흐름은 잘 구현했어.

가장 먼저 확인할 건 이 두 개야.

```python
# 1. 문제 좌표가 정말 maps[y][x] 방식이 맞는지
for i in range(c1, c2):
    for j in range(r1, r2):
        maps[i][j] = num

# 2. 점수 계산을 set 기반으로 단순화할지
record_res()
```

나라면 이 코드에서 제일 먼저 `recode_res()`를 set 기반으로 바꾸고, 그다음 변수명을 `i, j`보다 `y, x`로 정리할 것 같아. 지금 코드는 로직보다 좌표 해석에서 실수할 가능성이 더 커 보여.

### 💡 답변 (Assistant)
나라면 **문제 요구사항을 그대로 함수 단위로 쪼갰을 것 같아.**

핵심은 이렇게 나눠.

| 문제 요구사항 | 코드 모듈 |
|---|---|
| 미생물 투입 | `insert_microbe()` |
| 현재 배지에서 무리 찾기 | `find_groups()` |
| 쪼개진 무리 삭제 | `remove_split_groups()` |
| 새 배지로 이동 | `relocate_groups()` |
| 실험 결과 계산 | `calculate_score()` |

그리고 좌표는 명확하게 `board[y][x]`로 통일하는 게 좋아.  
네 코드도 사실상 이 방식이었어.

```python
from collections import Counter, deque
from dataclasses import dataclass


DIRECTIONS = [(-1, 0), (1, 0), (0, -1), (0, 1)]


@dataclass
class Group:
    microbe_id: int
    cells: list[tuple[int, int]]  # (y, x)

    @property
    def area(self) -> int:
        return len(self.cells)

    def normalized_cells(self) -> list[tuple[int, int]]:
        min_y = min(y for y, _ in self.cells)
        min_x = min(x for _, x in self.cells)

        return [(y - min_y, x - min_x) for y, x in self.cells]


def insert_microbe(
    board: list[list[int]],
    microbe_id: int,
    x1: int,
    y1: int,
    x2: int,
    y2: int,
) -> None:
    for y in range(y1, y2):
        for x in range(x1, x2):
            board[y][x] = microbe_id


def find_groups(board: list[list[int]], n: int) -> list[Group]:
    visited = [[False] * n for _ in range(n)]
    groups: list[Group] = []

    for sy in range(n):
        for sx in range(n):
            if board[sy][sx] == 0 or visited[sy][sx]:
                continue

            microbe_id = board[sy][sx]
            cells = bfs_group(board, visited, n, sy, sx, microbe_id)
            groups.append(Group(microbe_id, cells))

    return groups


def bfs_group(
    board: list[list[int]],
    visited: list[list[bool]],
    n: int,
    sy: int,
    sx: int,
    microbe_id: int,
) -> list[tuple[int, int]]:
    q = deque([(sy, sx)])
    visited[sy][sx] = True
    cells = [(sy, sx)]

    while q:
        y, x = q.popleft()

        for dy, dx in DIRECTIONS:
            ny, nx = y + dy, x + dx

            if not (0 <= ny < n and 0 <= nx < n):
                continue

            if visited[ny][nx]:
                continue

            if board[ny][nx] != microbe_id:
                continue

            visited[ny][nx] = True
            q.append((ny, nx))
            cells.append((ny, nx))

    return cells


def remove_split_groups(groups: list[Group]) -> list[Group]:
    group_count = Counter(group.microbe_id for group in groups)

    return [
        group
        for group in groups
        if group_count[group.microbe_id] == 1
    ]


def relocate_groups(groups: list[Group], n: int) -> list[list[int]]:
    new_board = [[0] * n for _ in range(n)]

    groups.sort(key=lambda group: (-group.area, group.microbe_id))

    for group in groups:
        cells = group.normalized_cells()
        position = find_position(new_board, cells, n)

        if position is None:
            continue

        base_y, base_x = position

        for dy, dx in cells:
            new_board[base_y + dy][base_x + dx] = group.microbe_id

    return new_board


def find_position(
    board: list[list[int]],
    cells: list[tuple[int, int]],
    n: int,
) -> tuple[int, int] | None:
    for base_x in range(n):
        for base_y in range(n):
            if can_place(board, cells, n, base_y, base_x):
                return base_y, base_x

    return None


def can_place(
    board: list[list[int]],
    cells: list[tuple[int, int]],
    n: int,
    base_y: int,
    base_x: int,
) -> bool:
    for dy, dx in cells:
        ny = base_y + dy
        nx = base_x + dx

        if not (0 <= ny < n and 0 <= nx < n):
            return False

        if board[ny][nx] != 0:
            return False

    return True


def calculate_score(board: list[list[int]], n: int) -> int:
    area = Counter()

    for y in range(n):
        for x in range(n):
            if board[y][x] > 0:
                area[board[y][x]] += 1

    adjacent_pairs = set()

    for y in range(n):
        for x in range(n):
            if board[y][x] == 0:
                continue

            current = board[y][x]

            for dy, dx in [(1, 0), (0, 1)]:
                ny, nx = y + dy, x + dx

                if not (0 <= ny < n and 0 <= nx < n):
                    continue

                other = board[ny][nx]

                if other > 0 and other != current:
                    adjacent_pairs.add(tuple(sorted((current, other))))

    result = 0

    for a, b in adjacent_pairs:
        result += area[a] * area[b]

    return result


def run_experiment(
    board: list[list[int]],
    n: int,
    microbe_id: int,
    command: list[int],
) -> int:
    x1, y1, x2, y2 = command

    insert_microbe(board, microbe_id, x1, y1, x2, y2)

    groups = find_groups(board, n)
    groups = remove_split_groups(groups)

    board[:] = relocate_groups(groups, n)

    return calculate_score(board, n)


def main() -> None:
    n, q = map(int, input().split())
    board = [[0] * n for _ in range(n)]

    for microbe_id in range(1, q + 1):
        command = list(map(int, input().split()))
        result = run_experiment(board, n, microbe_id, command)
        print(result)


if __name__ == "__main__":
    main()
```

내가 이렇게 짠 이유는 간단해.  
`run_experiment()`만 보면 문제 흐름이 그대로 보여.

```python
insert_microbe()
find_groups()
remove_split_groups()
relocate_groups()
calculate_score()
```

이 구조가 디버깅하기 제일 좋아. 예를 들어 점수가 틀리면 `calculate_score()`만 보면 되고, 이동이 틀리면 `relocate_groups()`랑 `find_position()`만 보면 돼.

주의할 점은 하나야. 이 코드는 입력 좌표를 `(x1, y1, x2, y2)`처럼 해석해. 즉 내부 저장은 `board[y][x]`야. 네 기존 코드의 `maps[i][j] = num` 방식과 같은 해석이야.

### 💡 답변 (Assistant)
좋아. 이 코드는 문제 요구사항을 그대로 함수로 쪼갠 구조야.  
핵심 관점은 **“한 번의 실험을 `run_experiment()`가 수행하고, 세부 단계는 각각 독립 함수가 맡는다”**야.

**전체 흐름**

```python
def run_experiment(board, n, microbe_id, command):
    x1, y1, x2, y2 = command

    insert_microbe(board, microbe_id, x1, y1, x2, y2)

    groups = find_groups(board, n)
    groups = remove_split_groups(groups)

    board[:] = relocate_groups(groups, n)

    return calculate_score(board, n)
```

이 함수가 문제의 한 턴을 그대로 표현해.

```text
1. 미생물 투입
2. 현재 배지에서 무리 찾기
3. 쪼개진 무리 제거
4. 새 배지로 이동
5. 결과 점수 계산
```

---

**1. `Group` 클래스**

```python
@dataclass
class Group:
    microbe_id: int
    cells: list[tuple[int, int]]
```

하나의 미생물 무리를 표현하는 객체야.

예를 들어 3번 미생물이 다음 칸들을 차지하고 있다면:

```python
Group(
    microbe_id=3,
    cells=[(1, 2), (1, 3), (2, 2)]
)
```

여기서 좌표는 전부 `(y, x)`야.  
즉 `board[y][x]` 기준으로 저장해.

```python
@property
def area(self) -> int:
    return len(self.cells)
```

`group.area`라고 쓰면 그 무리의 면적을 바로 얻을 수 있어.

```python
def normalized_cells(self):
    min_y = min(y for y, _ in self.cells)
    min_x = min(x for _, x in self.cells)

    return [(y - min_y, x - min_x) for y, x in self.cells]
```

이 함수가 중요해.  
미생물을 새 배지로 옮길 때는 **기존 모양은 유지하되 위치만 옮겨야 하잖아.**

예를 들어 원래 좌표가:

```python
[(3, 4), (3, 5), (4, 4)]
```

이면 최소 좌표 `(3, 4)`를 빼서:

```python
[(0, 0), (0, 1), (1, 0)]
```

으로 바꿔.  
이렇게 해야 새 배지의 어떤 기준점 `(base_y, base_x)`에 쉽게 배치할 수 있어.

---

**2. 미생물 투입**

```python
def insert_microbe(board, microbe_id, x1, y1, x2, y2):
    for y in range(y1, y2):
        for x in range(x1, x2):
            board[y][x] = microbe_id
```

직사각형 영역에 새 미생물을 넣는 함수야.

기존 미생물이 있어도 그냥 덮어써.

```python
board[y][x] = microbe_id
```

이게 곧 “새 미생물이 기존 미생물을 잡아먹는다”는 의미가 돼.

주의할 점은 입력을 `(x1, y1, x2, y2)`로 해석한다는 거야.  
문제에서 `(r1, c1, r2, c2)`라고 해도, 좌표 설명이 “좌측 하단, 우측 상단”이면 보통 `r`을 x, `c`를 y처럼 쓰는 문제들이 있어. 그래서 내부에서는 `board[y][x]`로 통일한 거야.

---

**3. 현재 배지에서 무리 찾기**

```python
def find_groups(board, n):
    visited = [[False] * n for _ in range(n)]
    groups = []
```

방문 배열을 만들고, 전체 배지를 훑어.

```python
for sy in range(n):
    for sx in range(n):
        if board[sy][sx] == 0 or visited[sy][sx]:
            continue
```

빈 칸이거나 이미 방문한 칸이면 넘어가.

```python
microbe_id = board[sy][sx]
cells = bfs_group(board, visited, n, sy, sx, microbe_id)
groups.append(Group(microbe_id, cells))
```

미방문 미생물을 발견하면 BFS로 같은 번호끼리 연결된 영역을 전부 찾고, `Group`으로 저장해.

**4. BFS로 하나의 무리 찾기**

```python
def bfs_group(board, visited, n, sy, sx, microbe_id):
    q = deque([(sy, sx)])
    visited[sy][sx] = True
    cells = [(sy, sx)]
```

시작 칸을 큐에 넣고 방문 처리해.

```python
for dy, dx in DIRECTIONS:
    ny, nx = y + dy, x + dx
```

상하좌우를 확인해.

```python
if board[ny][nx] != microbe_id:
    continue
```

같은 미생물 번호인 칸만 같은 무리로 인정해.

결과적으로 `bfs_group()`은 이런 리스트를 반환해.

```python
[(1, 2), (1, 3), (2, 2)]
```

즉, 한 무리가 차지하는 모든 좌표야.

**5. 쪼개진 무리 제거**

```python
def remove_split_groups(groups):
    group_count = Counter(group.microbe_id for group in groups)
```

각 미생물 번호가 몇 개의 그룹으로 나뉘었는지 세어.

예를 들어:

```python
[
    Group(1, ...),
    Group(2, ...),
    Group(2, ...),
    Group(3, ...)
]
```

이면 `2번 미생물`은 두 덩어리로 나뉜 상태야.

```python
return [
    group
    for group in groups
    if group_count[group.microbe_id] == 1
]
```

한 번호가 2개 이상의 그룹으로 나뉘었으면 전부 제거해.  
문제의 이 조건을 구현한 부분이야.

> A무리가 차지한 영역이 둘 이상 나눠지게 되면 나눠진 A무리의 미생물은 모두 사라짐.

---

**6. 새 배지로 이동**

```python
def relocate_groups(groups, n):
    new_board = [[0] * n for _ in range(n)]
```

새로운 빈 배지를 만들어.

```python
groups.sort(key=lambda group: (-group.area, group.microbe_id))
```

이동 우선순위대로 정렬해.

```text
1순위: 면적 큰 무리
2순위: 먼저 투입된 무리, 즉 microbe_id 작은 무리
```

그래서 `(-group.area, group.microbe_id)`가 돼.

```python
for group in groups:
    cells = group.normalized_cells()
    position = find_position(new_board, cells, n)
```

각 그룹의 모양을 `(0, 0)` 기준 상대 좌표로 바꾸고, 새 배지에서 놓을 수 있는 가장 앞 위치를 찾는다.

```python
if position is None:
    continue
```

놓을 수 없으면 사라짐.

```python
base_y, base_x = position

for dy, dx in cells:
    new_board[base_y + dy][base_x + dx] = group.microbe_id
```

놓을 수 있으면 새 배지에 배치해.

**7. 배치 위치 찾기**

```python
def find_position(board, cells, n):
    for base_x in range(n):
        for base_y in range(n):
            if can_place(board, cells, n, base_y, base_x):
                return base_y, base_x
```

여기서 순서가 중요해.

```python
for base_x in range(n):
    for base_y in range(n):
```

즉, **x좌표가 작은 위치를 먼저 찾고**, 같은 x라면 **y좌표가 작은 위치**를 찾는다.

문제 조건:

```text
최대한 x좌표가 작은 위치
그런 위치가 둘 이상이면 y좌표가 작은 위치
```

를 그대로 반영한 부분이야.

**8. 실제 배치 가능 여부 확인**

```python
def can_place(board, cells, n, base_y, base_x):
    for dy, dx in cells:
        ny = base_y + dy
        nx = base_x + dx
```

상대 좌표를 실제 좌표로 바꿔.

```python
if not (0 <= ny < n and 0 <= nx < n):
    return False
```

배지를 벗어나면 실패.

```python
if board[ny][nx] != 0:
    return False
```

이미 다른 미생물이 있으면 실패.

모든 칸이 통과하면:

```python
return True
```

**9. 점수 계산**

```python
def calculate_score(board, n):
    area = Counter()
```

먼저 각 미생물의 면적을 센다.

```python
for y in range(n):
    for x in range(n):
        if board[y][x] > 0:
            area[board[y][x]] += 1
```

예를 들어:

```python
area[1] = 4
area[2] = 3
area[5] = 7
```

이런 식으로 저장돼.

그다음 인접한 미생물 쌍을 찾는다.

```python
adjacent_pairs = set()
```

`set`을 쓰는 이유는 같은 두 무리가 여러 면에서 맞닿아도 한 번만 계산해야 하기 때문이야.

```python
for dy, dx in [(1, 0), (0, 1)]:
```

아래쪽, 오른쪽만 확인해.  
상하좌우를 전부 보면 같은 경계가 중복으로 잡힐 수 있어서, 오른쪽과 아래쪽만 보면 충분해.

```python
if other > 0 and other != current:
    adjacent_pairs.add(tuple(sorted((current, other))))
```

서로 다른 미생물이 맞닿아 있으면 쌍으로 저장해.

예를 들어 `(3, 1)`을 발견해도:

```python
tuple(sorted((3, 1)))  # (1, 3)
```

으로 저장하니까 `(1, 3)`과 `(3, 1)`이 중복되지 않아.

마지막으로 점수 계산.

```python
for a, b in adjacent_pairs:
    result += area[a] * area[b]
```

문제 조건 그대로:

```text
A 영역 * B 영역
```

을 모든 인접 쌍에 대해 더해.

---

**10. `board[:] = relocate_groups(...)` 의미**

```python
board[:] = relocate_groups(groups, n)
```

이건 기존 `board` 객체의 내용을 새 배지로 교체하는 코드야.

그냥 이렇게 해도 대부분 문제에서는 돼.

```python
board = relocate_groups(groups, n)
```

하지만 `run_experiment()` 바깥의 `board`까지 유지해서 바꾸려면 `board[:] = ...`가 안전해.  
리스트 객체 자체는 유지하고 내부 내용만 바꾸는 방식이야.

---

요약하면, 이 코드는 이렇게 읽으면 돼.

```python
main()
    매 실험마다 run_experiment()

run_experiment()
    투입
    그룹 탐색
    쪼개진 그룹 제거
    새 배지 이동
    점수 계산
```

그리고 가장 중요한 설계 포인트는 이거야.

```python
Group = 미생물 번호 + 좌표 목록
```

미생물을 숫자로만 다루지 않고, “하나의 무리”라는 단위로 묶었기 때문에 문제 요구사항을 함수로 나누기 쉬워진 거야.

### ❓ 질문 (User)

```
import sys
# sys.stdin = open('input.txt', 'r')
from typing import *

# ====================
"""
미생물 연구
N*N 정사각형 크기의 배지 

1. 미생물 투입
- 좌측 하단 좌표 (r1, c1), 우측 상단 좌표 (r2, c2) 직사각형 영역에 한 무리의 미생물 투입
- 만약 영역 내에 다른 미생물이 존재한다면 새로 투입된 미생물이 그 영역 내의 미생물들을 잡아먹음.
- 기존에 있던 어떤 미생물 무리 A가 새로 투입된 미생물 무리 B에게 잡아먹히면서 A무리가 차지한 영역이
    둘 이상 나눠지게 되면 나눠진 A무리의 미생물은 배지에서 모두 사라짐.

2. 배지 이동
- 모든 미생물을 새로운 배지로 이동. 새로운 배지의 크기도 기존 용기와 동일. 
- 이 과정은 기존 배지에 미생물이 한 마리도 존재하지 않을 떄 까지 다음 작업을 반복
- 기존 배지에 있는 무리 중 가장 차지한 영역이 넓은 무리를 하나 선택, 
- 만약 이런 미생물 무리가 둘 이상이면 먼저 투입된 미생물 선택
- 우선순위 : 크기 > idx
- 선택된 미생물 무리를 새 배지로 옮김. 
- 이 때, 무리는 기존 용기에서의 형태를 유지해야 함.
- 미생물 무리가 차지한 영역이 배양 용기의 범위를 벗어나지 않고, 
- 다른 미생물의 영역과 겹치지 않도록 두어야 함.
- 이 조건 안에서 최대한 x좌표가 작은 위치로 미생물을 옮겨야 함. 
- 그런 위치가 둘 이상일 경우 최대한 y좌표가 작은 위치로 옮겨야 함.
- 우선순위 : x좌표 > y좌표
- 만약 어떤 곳에도 둘 수 없는 미생물 무리가 발견되면, 그 미생물 무리는 사라짐.

3. 실험 결과 기록
- 미생물 무리 중 상하좌우로 맞닿은 면이 있는 무리끼리는 '인접한 무리' 라고 표현함.
- 모든 '인접한 무리' 쌍을 확인. 이때 두 무리 A와 B가 맞닿은 면이 둘 이상이더라도 (A, B) 쌍은 한 번만 확인
- 확인한 두 무리가 A, B라면 (A영역) * (B영역) 만큼의 성과를 얻음.
- 확인한 모든 쌍의 성과를 더한 값이 이 실험의 결과. 이 값을 기록

    return board

def count_cells_by_bfs(micro: Micro) :

    sr, sc = list(micro.cells.keys())[0]
    queue = deque()
    queue.append([sr, sc])
    visited.add((sr, sc))
    
    while queue :
        cr, cc = queue.popleft()

        for d in range(4) :
            nr, nc = cr + dr[d], cc + dc[d]
            if (nr, nc) not in visited and (nr, nc) in micro.cells.keys() :
                queue.append([nr, nc])

    return len(visited)
```

### 💡 답변 (Assistant)
두 코드의 핵심 구조는 거의 같아.  
다만 **두 번째 코드가 첫 번째 코드의 몇 가지 치명적인 버그를 고친 버전**에 가깝다. 특히 BFS 방문 처리, 이동 후 좌표 갱신, 인접 쌍 중복 처리에서 차이가 커.

**큰 결론부터 말하면**

두 번째 코드가 더 맞는 방향이야.  
첫 번째 코드는 아래 3개 때문에 오답 가능성이 높아.

1. `count_cells_by_bfs()`에서 방문 처리를 제대로 안 함
2. `place_board()`에서 `micro.c1`이 아니라 `micro.sc`로 오타
3. 인접 쌍 계산에서 같은 쌍을 두 번 더할 수 있음

**1. 입력 좌표 처리 차이**

첫 번째 코드:

```python
r1, c1, r2, c2 = map(int, input().split())
r1, c1 = r1 + 1, c1 + 1
micro = Micro(bios, r1, c1, r2, c2)
```

두 번째 코드:

```python
c1, r1, c2, r2 = map(int, input().split())
r1, c1 = r1 + 1, c1 + 1
micro = Micro(n, r1, c1, r2, c2)
```

두 번째는 입력을 사실상 `(x1, y1, x2, y2)`로 보고 있어.

즉 문제 입력이:

```text
x1 y1 x2 y2
```

라면 두 번째 코드가 더 자연스러워.

왜냐하면 내부 board는 이렇게 쓰고 있기 때문이야.

```python
board[row][col]
board[r][c]
```

그래서 입력의 x좌표는 `c`, y좌표는 `r`로 바꿔서 넣는 게 맞아.

```python
c1, r1, c2, r2 = map(int, input().split())
```

이 부분은 “x, y를 row, col로 바꾸는 과정”이라고 보면 돼.

---

**2. `count_cells_by_bfs()` 차이**

첫 번째 코드:

```python
if (nr, nc) not in visited and (nr, nc) in micro.cells.keys() :
    queue.append([nr, nc])
```

여기서 문제가 있어.  
큐에 넣은 다음 방문 처리를 안 하고 있어.

즉 아래 코드가 빠져 있어.

```python
visited.add((nr, nc))
```

그래서 첫 번째 코드는 연결된 칸 개수를 제대로 못 셀 수 있어. 심하면 같은 칸이 계속 큐에 들어가면서 이상하게 동작할 수도 있어.

두 번째 코드:

```python
if (nr, nc) not in visited and (nr, nc) in micro.cells.keys():
    queue.append([nr, nc])
    visited.add((nr, nc))
```

이게 맞아.

BFS에서는 보통 **큐에 넣는 순간 방문 처리**하는 게 안전해.

```python
queue.append(...)
visited.add(...)
```

이 패턴 기억해두면 좋아.

**3. 이동 후 위치 갱신 차이**

첫 번째 코드:


여기 오타가 있어.


하지만 `Micro` 클래스에는 `sc`라는 속성이 없어.  
원래 있어야 하는 건 `c1`이야.

즉 이렇게 되어야 해.

```python
micro.c1 = sc
```

두 번째 코드에서는 이게 고쳐져 있어.

```python
def place_board(board, sr, sc, micro):
```

```python
sr, sc = r + micro.r1, c + micro.c1
```

이렇게 현재 미생물의 위치를 기준으로 실제 board 좌표를 계산하거든.

**4. 인접 쌍 중복 처리 차이**

첫 번째 코드:

```python
for other_adj in local_adj :
    if (other_adj, micro.n) not in adj :
        adj.add((other_adj, micro.n))
```

이건 중복 제거가 잘 안 돼.

예를 들어 1번과 2번이 인접해 있다고 해보자.

1번 기준으로 보면:

```python
adj.add((2, 1))
```

2번 기준으로 보면:

```python
(1, 2) not in adj
adj.add((1, 2))
```

결국 `(2, 1)`과 `(1, 2)`가 둘 다 들어간다.  
그러면 점수를 두 번 더하게 돼.

두 번째 코드:

```python
for other_adjacent in local_adjacents:
    if (other_adjacent, micro.n) not in adjacents:
        adjacents.add((micro.n, other_adjacent))
```

이건 동작이 더 좋아.

1번 기준으로 2번을 찾으면:

```python
adjacents.add((1, 2))
```

2번 기준으로 1번을 찾으면:

```python
if (1, 2) not in adjacents:
```

이미 있으니까 추가하지 않아.

그래서 두 번째 코드가 중복 방지를 제대로 하고 있어.

다만 더 깔끔하게 하려면 이렇게 하는 게 제일 안전해.

```python
pair = tuple(sorted((micro.n, other_adjacent)))
adjacents.add(pair)
```

그러면 굳이 reverse 체크를 하지 않아도 돼.

---

**5. `tetris_micro()`는 두 코드가 거의 같음**

핵심은 이 부분이야.

```python
heapq.heappush(candidates, (c, r))
```

후보 위치를 `(c, r)`로 넣고 있어.

그러면 heap은 자동으로:

```text
c가 작은 순
c가 같으면 r이 작은 순
```

으로 꺼내.

문제 조건이:

```text
x좌표가 작은 위치 우선
그다음 y좌표가 작은 위치
```

라면 `c`를 x좌표로 보고 있으므로 맞는 방향이야.

다만 현재 방식은 모든 후보를 heap에 넣은 뒤 하나를 꺼내고 있어.

```python
for r in range(N - 1, 0, -1):
    for c in range(1, N):
        if can_place_micro(...):
            heapq.heappush(candidates, (c, r))
```

동작은 가능하지만, 꼭 heap이 필요하진 않아.  
어차피 우선순위가 명확하니까 처음부터 순서대로 탐색해도 돼.

```python
for c in range(1, N):
    for r in range(1, N):
        if can_place_micro(board, r, c, micro):
            place_board(board, r, c, micro)
            return True
```

이렇게 하면 `heapq`가 필요 없어지고 더 단순해져.

**6. 첫 번째 코드의 또 다른 자잘한 문제**

첫 번째 코드에는 이 부분이 있어.

```python
micro_order_heap = []
```

이게 `for bios in range(...)` 바깥에 있어.

하지만 매 실험 끝에 `while micro_order_heap:`로 전부 비우기 때문에 실제로 큰 문제는 안 생길 가능성이 높아.  
그래도 구조상으로는 매 실험마다 새로 만드는 게 더 안전해.

```python
for n in range(1, Q + 1):
    micro_order_heap = []
```

두 번째 코드도 현재는 바깥에 있지만, 마찬가지로 안쪽으로 넣는 편이 더 좋다.

**7. 자료구조 관점**

두 코드 모두 `Micro.cells`를 dict로 쓰고 있어.

```python
self.cells = {}
self.cells[(nr, nc)] = 0
```

그런데 value인 `0`은 전혀 쓰지 않아.  
이 경우는 `set`이 더 자연스러워.

```python
self.cells = set()
self.cells.add((nr, nc))
```

그러면 이런 코드도 더 자연스럽게 바뀐다.

```python
if (nr, nc) in micro.cells:
```

지금 dict를 써도 동작은 하지만, 의미상으로는 set이 맞아.

---

**최종 평가**

| 항목 | 첫 번째 코드 | 두 번째 코드 |
|---|---:|---:|
| 좌표 변환 | 애매함 | 더 명확함 |
| BFS 연결 개수 계산 | 버그 있음 | 수정됨 |
| 이동 후 위치 갱신 | `micro.sc` 오타 | `micro.c1` 정상 |
| 인접 쌍 중복 방지 | 중복 가능 | 개선됨 |
| 전체 구조 | 동작 불안정 | 더 안정적 |

내 기준으로는 **두 번째 코드가 확실히 더 낫다.**  
첫 번째 코드는 BFS 방문 처리랑 `micro.sc` 오타 때문에 결과가 크게 틀어질 수 있어.

다만 두 번째 코드도 더 개선한다면 이 세 가지를 추천해.

```python
# 1. 인접 쌍은 sorted tuple로 관리
pair = tuple(sorted((micro.n, other_adjacent)))
adjacents.add(pair)

# 2. micro_order_heap은 매 실험마다 새로 생성
micro_order_heap = []

# 3. cells는 dict보다 set 사용
self.cells = set()
```

그리고 제일 중요한 건 이거야.  
이 문제는 로직보다 **좌표계가 더 위험한 문제**야. 내부에서 `board[r][c]`를 쓰는 이상, 입력이 `(x1, y1, x2, y2)`인지 `(r1, c1, r2, c2)`인지 먼저 확정해야 해. 두 번째 코드는 입력을 `(x1, y1, x2, y2)`로 해석한 코드라고 보면 돼.
