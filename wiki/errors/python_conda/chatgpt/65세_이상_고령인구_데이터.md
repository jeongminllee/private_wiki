### ✅ 로컬에서 실행 가능한 코드 예시

```python
import pandas as pd
import geopandas as gpd
import matplotlib.pyplot as plt
import json
from shapely.geometry import shape

# 1. 데이터 불러오기
df_elderly = pd.read_excel("등록인구_20250519214713.xlsx", sheet_name='데이터', skiprows=2)
df_elderly = df_elderly[['동별(2)', '소계.5']].dropna()
df_elderly.columns = ['자치구', '65세 이상 고령인구수']
df_elderly = df_elderly[df_elderly['자치구'] != '소계']
df_elderly['65세 이상 고령인구수'] = pd.to_numeric(df_elderly['65세 이상 고령인구수'])

# 2. GeoJSON 불러오기
with open("seoul_municipalities_geo_simple.json", "r", encoding="utf-8") as f:
    geojson_data = json.load(f)

# 3. GeoDataFrame 생성
features = []
for feature in geojson_data['features']:
    name = feature['properties']['name']
    geom = shape(feature['geometry'])
    features.append({'자치구': name, 'geometry': geom})
geo_df = gpd.GeoDataFrame(features)

# 4. 병합 및 시각화
merged = geo_df.merge(df_elderly, on='자치구', how='left')
merged.plot(column='65세 이상 고령인구수', cmap='OrRd', legend=True, edgecolor='black', figsize=(12, 12))
plt.title("서울시 자치구별 65세 이상 고령인구수 분포", fontsize=16)
plt.axis('off')
plt.tight_layout()
plt.show()
```

## ✅ 해결 방법: 로컬에서 실행 가능한 Folium 코드 (인터랙티브 지도 생성)

아래 코드를 복사하여 Jupyter Notebook이나 Python IDE에서 실행하세요.

```python
import pandas as pd
import geopandas as gpd
import folium
from folium import Choropleth
from shapely.geometry import shape, mapping
import json

# 1. 고령 인구 데이터 로드
df = pd.read_excel("등록인구_20250519214713.xlsx", sheet_name='데이터', skiprows=2)
df_elderly = df[['동별(2)', '소계.5']].dropna()
df_elderly.columns = ['자치구', '65세 이상 고령인구수']
df_elderly = df_elderly[df_elderly['자치구'] != '소계']
df_elderly['65세 이상 고령인구수'] = pd.to_numeric(df_elderly['65세 이상 고령인구수'])

# 2. GeoJSON 로드 및 geometry 변환
with open("seoul_municipalities_geo_simple.json", "r", encoding="utf-8") as f:
    geojson_data = json.load(f)

for feature in geojson_data['features']:
    feature['geometry'] = mapping(shape(feature['geometry']))  # shapely → dict

# 3. Folium 지도 생성
seoul_center = [37.5665, 126.9780]
m = folium.Map(location=seoul_center, zoom_start=11)

# 4. Choropleth 레이어 추가
Choropleth(
    geo_data=geojson_data,
    name="65세 이상 고령인구수",
    data=df_elderly,
    columns=["자치구", "65세 이상 고령인구수"],
    key_on="feature.properties.name",
    fill_color="YlOrRd",
    fill_opacity=0.7,
    line_opacity=0.2,
    legend_name="65세 이상 고령인구수",
).add_to(m)

# 5. HTML로 저장
m.save("seoul_elderly_population_map.html")
```

import pandas as pd
import folium
from folium import Choropleth
import json
from shapely.geometry import shape, mapping

# 1. 인구 데이터
df = pd.read_excel("등록인구_20250519214713.xlsx", sheet_name='데이터', skiprows=2)
